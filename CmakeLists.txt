cmake_minimum_required(VERSION 3.22)

project(nucleo_hal
    LANGUAGES C CXX ASM
)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Resolve the preset's BOARD to the BSP directory and its MCU configuration.
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/boards.cmake")
include("${BSP_BOARD_DIR}/target.cmake")
message(STATUS "BSP board: ${BSP_BOARD}; MCU: ${BSP_MCU}")

# ---------------------------------------------------------------------------
# Automatically collect CubeMX-generated sources.
# ---------------------------------------------------------------------------

file(GLOB_RECURSE MCAL_CORE_SOURCES CONFIGURE_DEPENDS
    "${MCAL_DIR}/Core/Src/*.c"
)

# appl/src/main.cpp is the real application entry point.
list(FILTER MCAL_CORE_SOURCES EXCLUDE REGEX
    "[/\\\\]main\\.c$"
)

file(GLOB_RECURSE MCAL_DRIVER_SOURCES CONFIGURE_DEPENDS
    "${MCAL_DRIVER_DIR}/Src/*.c"
)

# ---------------------------------------------------------------------------
# Final firmware executable.
# ---------------------------------------------------------------------------

add_executable(${PROJECT_NAME}
    ${MCAL_CORE_SOURCES}
    ${MCAL_DRIVER_SOURCES}
    "${MCAL_STARTUP_SOURCE}"
)

set_target_properties(${PROJECT_NAME} PROPERTIES
    SUFFIX ".elf"
)

add_subdirectory(bsp)
add_subdirectory(appl)

target_link_libraries(${PROJECT_NAME} PRIVATE
    mcal_config
    bsp_board
    m
)

target_compile_options(${PROJECT_NAME} PRIVATE
    -Wall

    $<$<COMPILE_LANGUAGE:CXX>:-Wextra>
    $<$<COMPILE_LANGUAGE:CXX>:-Wpedantic>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-threadsafe-statics>
)

# ---------------------------------------------------------------------------
# Create a linker-script copy compatible with older GNU Arm linkers.
#
# New CubeMX versions may generate output sections using:
#
#     .ARM.extab (READONLY) :
#
# Older GNU ld versions do not recognize the READONLY section attribute.
# Keep the generated mcal linker script unchanged and patch a build copy.
# ---------------------------------------------------------------------------

set(COMPATIBLE_LINKER_SCRIPT
    "${CMAKE_BINARY_DIR}/firmware_FLASH.ld"
)

# Reconfigure automatically when CubeMX regenerates the linker script.
set_property(
    DIRECTORY
    APPEND
    PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${GENERATED_LINKER_SCRIPT}"
)

file(
    READ
    "${GENERATED_LINKER_SCRIPT}"
    LINKER_SCRIPT_CONTENT
)

# The FLASH memory region already makes these sections read-only.
string(
    REPLACE
    "(READONLY)"
    ""
    LINKER_SCRIPT_CONTENT
    "${LINKER_SCRIPT_CONTENT}"
)

file(
    WRITE
    "${COMPATIBLE_LINKER_SCRIPT}"
    "${LINKER_SCRIPT_CONTENT}"
)

set_property(
    TARGET ${PROJECT_NAME}
    APPEND
    PROPERTY LINK_DEPENDS
    "${GENERATED_LINKER_SCRIPT}"
)

target_link_options(${PROJECT_NAME} PRIVATE
    ${MCU_ARCH_OPTIONS}

    --specs=nano.specs

    -T${COMPATIBLE_LINKER_SCRIPT}

    -Wl,-Map=${CMAKE_BINARY_DIR}/${PROJECT_NAME}.map
    -Wl,--gc-sections
    -Wl,--print-memory-usage
)

# Generate .hex and .bin files after linking.
if(CMAKE_OBJCOPY)
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
        COMMAND ${CMAKE_OBJCOPY}
            -O ihex
            $<TARGET_FILE:${PROJECT_NAME}>
            "${CMAKE_BINARY_DIR}/${PROJECT_NAME}.hex"

        COMMAND ${CMAKE_OBJCOPY}
            -O binary
            $<TARGET_FILE:${PROJECT_NAME}>
            "${CMAKE_BINARY_DIR}/${PROJECT_NAME}.bin"

        COMMENT "Generating HEX and BIN files"
    )
endif()

if(CMAKE_SIZE)
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
        COMMAND ${CMAKE_SIZE}
            $<TARGET_FILE:${PROJECT_NAME}>

        COMMENT "Firmware size"
    )
endif()
